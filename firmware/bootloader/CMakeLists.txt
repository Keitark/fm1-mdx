cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(fm1_boot_policy C)
  enable_testing()
endif()
add_library(fm1_boot_policy boot_policy.c)
target_include_directories(fm1_boot_policy PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(fm1_boot_policy PUBLIC c_std_11)
add_executable(boot_policy_tests test_boot_policy.c)
target_link_libraries(boot_policy_tests PRIVATE fm1_boot_policy)
add_test(NAME source_boot_policy COMMAND boot_policy_tests)
